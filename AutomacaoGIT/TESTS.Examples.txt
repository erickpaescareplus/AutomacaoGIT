    public class GitAutomationResultTests
    {
        [Fact]
        public void AddLog_ShouldAddTimestampedMessage()
        {
            // Arrange
            var result = new GitAutomationResult();
            var message = "Test message";

            // Act
            result.AddLog(message);

            // Assert
            result.Logs.Should().HaveCount(1);
            result.Logs[0].Should().Contain(message);
            result.Logs[0].Should().MatchRegex(@"\[\d{2}:\d{2}:\d{2}\]");
        }

        [Fact]
        public void AddError_ShouldAddToErrorsAndLogs()
        {
            // Arrange
            var result = new GitAutomationResult();
            var error = "Test error";

            // Act
            result.AddError(error);

            // Assert
            result.Errors.Should().HaveCount(1);
            result.Logs.Should().HaveCount(1);
            result.Errors[0].Should().Contain("ERRO");
            result.Errors[0].Should().Contain(error);
        }

        [Fact]
        public void IsSuccess_WhenStatusIsSuccess_ShouldReturnTrue()
        {
            // Arrange
            var result = new GitAutomationResult
            {
                Status = GitOperationStatus.Success
            };

            // Assert
            result.IsSuccess.Should().BeTrue();
        }

        [Fact]
        public void Duration_WithStartAndEndTime_ShouldCalculateCorrectly()
        {
            // Arrange
            var result = new GitAutomationResult
            {
                StartTime = DateTime.Now,
                EndTime = DateTime.Now.AddSeconds(5)
            };

            // Assert
            result.Duration.Should().NotBeNull();
            result.Duration.Value.TotalSeconds.Should().BeApproximately(5, 0.1);
        }
    }

 * COMANDOS ÚTEIS PARA TESTES:

 *   dotnet test --filter Category!=Integration

 * Executar com cobertura de código:
 *   dotnet test /p:CollectCoverage=true
 * 
 * Executar em modo verbose:
